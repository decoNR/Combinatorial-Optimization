\begin{algorithm}
  \caption{Verify-Directed-Cut($D$, $B$, $S$)}\label{dcut}
  \begin{algorithmic}
    \For{each $a \in A(D)$}
      \State $v,u \gets ends[a]$

      \If{$v \in S$ and $u \notin S$}
        \If{$a$ $\notin B$}
          \State \Return False
        \EndIf

      \ElsIf{$v \notin S$ and $u \in S$}
        \State \Return False
 

      \ElsIf{$v \in S$ and $u \in S$}
        \If{$a$ $\in B$}
          \State \Return False
        \EndIf

      \Else 
        \If{$a$ $\in B$}
          \State \Return False
        \EndIf
      \EndIf

    \EndFor

    \State \Return True
      
  \end{algorithmic}
\end{algorithm}
